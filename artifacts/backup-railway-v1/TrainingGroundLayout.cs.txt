        static Vector3 P(float x, float z) => new Vector3(x, 0, z);

        static List<Vector3> Arc(Vector3 center, float r, float angleStart, float angleEnd, int steps = 24)
        {
            var pts = new List<Vector3>();
            for (int i = 0; i <= steps; i++)
            {
                float a = Mathf.Lerp(angleStart, angleEnd, (float)i / steps) * Mathf.Deg2Rad;
                pts.Add(center + new Vector3(Mathf.Cos(a) * r, 0, Mathf.Sin(a) * r));
            }
            return pts;
        }

        static void SegmentPaint(Vector3 a, Vector3 b, Material mat, float thickness = .14f)
        {
            var delta = b - a;
            if (delta.magnitude < .005f) return;
            var ob = Box("Road edge", (a + b) * .5f + Vector3.up * .035f, new Vector3(thickness, .012f, delta.magnitude + .015f), mat, markings);
            ob.transform.rotation = Quaternion.LookRotation(delta.normalized);
        }

        static void Road(string name, IList<Vector3> path, float width, bool edges = true)
        {
            if (path.Count < 2) return;
            var vertices = new Vector3[path.Count * 2];
            var triangles = new List<int>();
            for (int i = 0; i < path.Count; i++)
            {
                Vector3 f;
                if (i == 0) f = (path[1] - path[0]).normalized;
                else if (i == path.Count - 1) f = (path[i] - path[i - 1]).normalized;
                else f = (path[i + 1] - path[i - 1]).normalized;

                var right = Vector3.Cross(Vector3.up, f).normalized;
                vertices[i * 2] = path[i] - right * width / 2 + Vector3.up * .015f;
                vertices[i * 2 + 1] = path[i] + right * width / 2 + Vector3.up * .015f;
                if (i > 0)
                {
                    int k = (i - 1) * 2;
                    triangles.AddRange(new[] { k, k + 2, k + 1, k + 1, k + 2, k + 3 });
                    if (edges)
                    {
                        SegmentPaint(vertices[k], vertices[k + 2], white);
                        SegmentPaint(vertices[k + 1], vertices[k + 3], white);
                    }
                }
            }
            var mesh = new Mesh { name = name, vertices = vertices, triangles = triangles.ToArray() };
            mesh.RecalculateNormals();
            string pathName = Base + "/Art/TrainingKit/" + name + ".asset";
            var old = AssetDatabase.LoadAssetAtPath<Mesh>(pathName);
            if (old) { EditorUtility.CopySerialized(mesh, old); UnityEngine.Object.DestroyImmediate(mesh); mesh = old; }
            else AssetDatabase.CreateAsset(mesh, pathName);
            var go = new GameObject(name, typeof(MeshFilter), typeof(MeshRenderer));
            go.transform.SetParent(environment);
            go.GetComponent<MeshFilter>().sharedMesh = mesh;
            go.GetComponent<MeshRenderer>().sharedMaterial = zoneAsphalt;
            go.layer = 9;
            go.isStatic = true;
            var mc = go.AddComponent<MeshCollider>();
            mc.sharedMesh = mesh;
        }

        static void Road(string name, float width, params Vector3[] points) => Road(name, (IList<Vector3>)points, width);

        static GameObject PlaceSign(string name, float x, float z, float yaw = 0)
        {
            string prefabPath = Base + "/Prefabs/Traffic/" + name + ".prefab";
            var asset = AssetDatabase.LoadAssetAtPath<GameObject>(prefabPath);
            if (!asset)
            {
                string fbxPath = Base + "/Art/Traffic/" + name + ".fbx";
                asset = AssetDatabase.LoadAssetAtPath<GameObject>(fbxPath);

            go.layer = 0;

        // One of each exercise, in metres; north is +Z. Reference footprint: 100 x 100.
        static List<Vector3> Rounded(float radius, params Vector3[] corners)
        {
            var points = new List<Vector3> { corners[0] };
            for (int i = 1; i < corners.Length - 1; i++)
            {
                var incoming = (corners[i] - corners[i - 1]).normalized;
                var outgoing = (corners[i + 1] - corners[i]).normalized;
                float r = Mathf.Min(radius, Mathf.Min(Vector3.Distance(corners[i], corners[i-1]), Vector3.Distance(corners[i], corners[i+1])) * .45f);
                var a = corners[i] - incoming * r;
                var b = corners[i] + outgoing * r;
                points.Add(a);
                for (int j = 1; j <= 16; j++)
                {
                    float t = j / 16f;
                    points.Add((1-t)*(1-t)*a + 2*(1-t)*t*corners[i] + t*t*b);
                }
            }
            points.Add(corners[corners.Length - 1]);
            return points;
        }
